      document.getElementById('buyFoodBtn')?.addEventListener('click', ()=>{
        state.showShop = !state.showShop;
        render();
      });
      document.getElementById('closeShopBtn')?.addEventListener('click', ()=>{
        state.showShop = false;
        render();
      });
      // 购买食物
      document.querySelectorAll('[data-foodidx]').forEach(btn => {
        btn.addEventListener('click', function() {
          const idx = parseInt(this.dataset.foodidx);
          const food = FOODS[idx];
          if (state.coins >= food.price) {
            state.coins -= food.price;
            state.foodInventory.push(food.name);
            state.hunger = Math.min(100, state.hunger + 10);
            alert(`✅ 买了 ${food.name}！ 小宠开心～`);
            render();
          } else {
            alert('😢 金币不够啦！ 摸摸它攒点爱吧～');
          }
        });
      });

      document.getElementById('feedBtn')?.addEventListener('click', ()=>{
        if (state.foodInventory.length > 0) {
          state.foodInventory.pop();
          state.hunger = Math.min(100, state.hunger + 15);
          state.happiness = Math.min(100, state.happiness + 5);
          alert('🍽️ 喂食成功！ 小家伙吃饱了～');
          render();
        } else {
          alert('🥺 没有食物了，去小卖部买点吧！');
        }
      });

      document.getElementById('petBtn')?.addEventListener('click', ()=>{
        state.happiness = Math.min(100, state.happiness + 12);
        state.hunger = Math.max(0, state.hunger - 4);
        alert('🤚 你轻轻摸了摸它，它呼噜呼噜～');
        render();
      });

      document.getElementById('exitBtn')?.addEventListener('click', ()=>{
        if (confirm('真的要退出吗？小宠会想你的 🥺')) {
          state.step = 'invite';
          state.selectedPetId = null;
          state.showShop = false;
          render();
        }
      });

      // 小纸条toggle
      document.querySelectorAll('.letter-badge').forEach(el => {
        el.addEventListener('click', function() {
          const idx = parseInt(this.dataset.letteridx);
          state.showLetter = (state.showLetter === idx) ? null : idx;
          render();
        });
      });
    }

    // 初始渲染
    render();
  })();
</script>
